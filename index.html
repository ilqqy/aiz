🔁 Рекурсия

Правило: (1) przypadek bazowy — когда ответ известен сразу; (2) krok rekurencyjny — вызов себя с меньшим аргументом и использование результата.
Как найти ① и ②: посчитай руками для 7, 23, 234. ① = ответ для одной цифры. ② = как ответ для 234 получается из ответа для 23 и цифры 4.
Шаблон: рекурсия по цифрам числа
static int f(int n) {
    if (n < 10) {
        return ①;                  // ответ для однозначного числа
    }
    int lastDigit = n % 10;         // 234 -> 4
    int rest = f(n / 10);           // ответ для 23
    return ②;                      // склеить lastDigit и rest
}
задача①②пример
сумма цифрnlastDigit + rest742 → 13
произведение цифрnlastDigit * rest234 → 24, 205 → 0, 0 → 0
количество цифр1rest + 1234 → 3, 0 → 1
максимальная цифраnMath.max(lastDigit, rest)274 → 7, 91 → 9
минимальная цифраnMath.min(lastDigit, rest)274 → 2
сколько чётных цифрn чётное → 1, иначе 0 (if/else)lastDigit чётная → rest + 1, иначе rest2741 → 2
Ловушки: f(n + 1) или f(n) → нет прогресса → StackOverflowError. f(n / 10); без использования результата → неверный ответ. return 5 вместо return n — n это любое число. «Рекурсивно» = без циклов и без String.
Silnia (factorial)
static long factorial(int n) {      // 0 <= n <= 20
    if (n == 0) {
        return 1;                    // 0! = 1
    }
    long child = factorial(n - 1);
    return n * child;                // 3! = 3 * 2! = 6
}
Fibonacci (рекурсивно, лекция 05)
static long fib(int n) {             // F(0)=0, F(1)=1, F(n)=F(n-1)+F(n-2)
    if (n <= 1) {
        return n;
    }
    long left = fib(n - 1);
    long right = fib(n - 2);
    return left + right;             // fib(10) = 55
}
Fibonacci циклом (если спросят «porównaj»)
static long fibIter(int n) {         // без рекурсии: Θ(n) czasu, O(1) pamięci
    if (n <= 1) {
        return n;
    }
    long a = 0;
    long b = 1;
    for (int i = 2; i <= n; i++) {
        long next = a + b;
        a = b;
        b = next;
    }
    return b;
}
Ещё: степень, сумма массива рекурсивно
static int power(int base, int exp) {     // base^exp, exp >= 0
    if (exp == 0) {
        return 1;
    }
    int rest = power(base, exp - 1);
    return base * rest;                    // power(2, 10) = 1024
}

static int sumArray(int[] a, int i) {     // вызов: sumArray(a, 0)
    if (i == a.length) {
        return 0;                          // пустой «хвост»
    }
    int rest = sumArray(a, i + 1);
    return a[i] + rest;
}

Каждый вызов имеет свою переменную n (ramka stosu). Вызовы идут вниз до базы, результаты поднимаются вверх. fib(4): 9 вызовов, но на стеке одновременно ≤ 4.

Insertion Sort (przez wstawianie)

static void insertionSort(int[] a) {
    for (int i = 1; i < a.length; i++) {      // берём карты со второй
        int key = a[i];                        // карта в руке
        int j = i - 1;
        while (j >= 0 && a[j] > key) {         // <- убывание: a[j] < key
            a[j + 1] = a[j];                   // сдвиг вправо (не swap!)
            j--;
        }
        a[j + 1] = key;                        // вставить в освободившееся место
    }
}
[4, 2, 5, 3]
key=2: 4>2 сдвиг             → [2, 4, 5, 3]
key=5: 4>5? нет              → [2, 4, 5, 3]
key=3: 5>3 сдвиг, 4>3 сдвиг, 2>3? нет → [2, 3, 4, 5]

Bubble Sort (bąbelkowe)

static void bubbleSort(int[] a) {
    for (int end = a.length - 1; end > 0; end--) {
        boolean changed = false;
        for (int j = 0; j < end; j++) {
            if (a[j] > a[j + 1]) {             // <- убывание: a[j] < a[j + 1]
                int t = a[j];
                a[j] = a[j + 1];
                a[j + 1] = t;
                changed = true;
            }
        }
        if (!changed) {
            break;                             // проход без замен -> готово
        }
    }
}
[4, 2, 5, 3]
проход 1: (4,2) swap → [2,4,5,3]; (4,5) ок; (5,3) swap → [2,4,3,5]   ← 5 на месте
проход 2: (2,4) ок; (4,3) swap → [2,3,4,5]
проход 3: замен нет → break
[3,1,2]: проход 1 → [1,2,3], проход 2 без замен → стоп  (= 2 прохода, не 1!)

Selection Sort (przez wybieranie)

static void selectionSort(int[] a) {
    for (int i = 0; i < a.length - 1; i++) {
        int min = i;                           // индекс минимума остатка
        for (int j = i + 1; j < a.length; j++) {
            if (a[j] < a[min]) {               // <- убывание: a[j] > a[min]
                min = j;
            }
        }
        int tmp = a[i];
        a[i] = a[min];
        a[min] = tmp;
    }
}
[4, 2, 5, 3]
i=0: минимум 2 → swap с 4 → [2, 4, 5, 3]
i=1: минимум 3 → swap с 4 → [2, 3, 5, 4]
i=2: минимум 4 → swap с 5 → [2, 3, 4, 5]
сравнений всегда n(n−1)/2 (для 4 элементов: 3+2+1 = 6)

QuickSort (podział Lomuto, лекция 06)

static void quickSort(int[] a, int l, int r) {     // вызов: quickSort(a, 0, a.length - 1)
    if (l >= r) {
        return;                                     // 0 или 1 элемент
    }
    int pivot = a[r];                               // pivot = последний
    int i = l;                                      // следующее место для "<= pivot"
    for (int j = l; j < r; j++) {
        if (a[j] <= pivot) {                        // <- убывание: a[j] >= pivot
            int t = a[i];
            a[i] = a[j];
            a[j] = t;
            i++;
        }
    }
    int t = a[i];                                   // pivot на окончательное место i
    a[i] = a[r];
    a[r] = t;
    quickSort(a, l, i - 1);
    quickSort(a, i + 1, r);
}
[5, 2, 7, 1, 4], pivot = 4, i = 0
j=0: 5 > 4 пропуск
j=1: 2 ≤ 4 swap a[0]↔a[1] → [2, 5, 7, 1, 4], i=1
j=2: 7 > 4 пропуск
j=3: 1 ≤ 4 swap a[1]↔a[3] → [2, 1, 7, 5, 4], i=2
pivot ↔ a[2] → [2, 1 | 4 | 5, 7]   → дальше сортируем [2,1] и [5,7]

Худший случай Θ(n²): уже отсортированные данные или все равные (одна часть пустая). Лучший/средний Θ(n log n).

MergeSort (scalanie)

static int[] merge(int[] a, int[] b) {
    int[] result = new int[a.length + b.length];
    int i = 0;
    int j = 0;
    int k = 0;
    while (i < a.length && j < b.length) {
        if (a[i] <= b[j]) {                // <= : при равенстве берём левый -> стабильно
            result[k] = a[i];
            i++;
        } else {
            result[k] = b[j];
            j++;
        }
        k++;
    }
    while (i < a.length) {                 // остаток копируем без сравнений
        result[k] = a[i];
        i++;
        k++;
    }
    while (j < b.length) {
        result[k] = b[j];
        j++;
        k++;
    }
    return result;
}

static int[] mergeSort(int[] a) {          // нужен import java.util.Arrays;
    if (a.length <= 1) {
        return a.clone();
    }
    int mid = a.length / 2;
    int[] left = mergeSort(Arrays.copyOfRange(a, 0, mid));
    int[] right = mergeSort(Arrays.copyOfRange(a, mid, a.length));
    return merge(left, right);
}
merge([2, 5], [1, 7]):  1 (из правой) → 2 → 5 (левая кончилась) → дописать 7 → [1, 2, 5, 7]
mergeSort([5,2,7,1]) → [5,2] [7,1] → [5][2][7][1] → [2,5] [1,7] → [1,2,5,7]

🔍 Поиск

Линейный поиск
static int linearSearch(int[] a, int x) {     // любые данные, O(n)
    for (int i = 0; i < a.length; i++) {
        if (a[i] == x) {
            return i;
        }
    }
    return -1;
}
Binary search (индекс или −1)
static int binarySearch(int[] a, int target) {   // ТОЛЬКО отсортированный массив, O(log n)
    int left = 0;
    int right = a.length - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;      // пересчитывать КАЖДЫЙ круг
        if (a[mid] == target) {
            return mid;
        } else if (a[mid] < target) {
            left = mid + 1;                       // середина мала -> вправо
        } else {
            right = mid - 1;                      // середина велика -> влево
        }
    }
    return -1;                                    // пустой массив тоже даст -1
}
[4, 9, 12, 18, 23, 31]   (индексы 0..5)
target 23: (L0 R5 mid2: 12<23 → L=3) → (L3 R5 mid4: 23==23) → 4
target 10: (L0 R5 mid2: 12>10 → R=1) → (L0 R1 mid0: 4<10 → L=1) → (L1 R1 mid1: 9<10 → L=2) → L>R → −1
Следующая строка = те же L,R, но ОДНА граница сдвинута на mid±1.

Массив по убыванию? Поменять местами left = mid + 1 и right = mid - 1 (т.е. условие a[mid] > target → вправо).

Позиция вставки / первое вхождение (lower bound)
static int insertPosition(int[] a, int x) {     // первая позиция с a[pos] >= x (LeetCode 35)
    int left = 0;
    int right = a.length;                        // [left, right) — right не входит
    while (left < right) {
        int mid = left + (right - left) / 2;
        if (a[mid] < x) {
            left = mid + 1;
        } else {
            right = mid;                         // равенство НЕ останавливает
        }
    }
    return left;                                 // [1,3,3,3,8]: x=3 -> 1, x=10 -> 5
}

Первое вхождение x: pos = insertPosition(a, x); если pos < a.length && a[pos] == x → pos, иначе −1.

public class ZnajdzNumer {

    // Wyszukiwanie binarne w tablicy uporządkowanej rosnąco; zwraca indeks albo -1.
    static int findNumber(int[] numbers, int target) {
        int left = 0;
        int right = numbers.length - 1;
        while (left <= right) {
            int mid = left + (right - left) / 2;
            if (numbers[mid] == target) {
                return mid;
            } else if (numbers[mid] < target) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        return -1;
    }

    public static void main(String[] args) {
        // przykłady z polecenia
        System.out.println(findNumber(new int[]{4, 9, 12, 18, 23, 31}, 12)); // 2
        System.out.println(findNumber(new int[]{4, 9, 12, 18, 23, 31}, 32)); // -1
        System.out.println(findNumber(new int[]{}, 12));                     // -1
        // przypadki brzegowe
        System.out.println(findNumber(new int[]{4, 9, 12, 18, 23, 31}, 4));  // 0
        System.out.println(findNumber(new int[]{4, 9, 12, 18, 23, 31}, 31)); // 5
        System.out.println(findNumber(new int[]{4, 9, 12, 18, 23, 31}, 1));  // -1
        System.out.println(findNumber(new int[]{4, 9, 12, 18, 23, 31}, 10)); // -1
        System.out.println(findNumber(new int[]{7}, 7));                     // 0
        System.out.println(findNumber(new int[]{7}, 8));                     // -1
    }
}

        if (n < 10) {

import java.util.Arrays;

public class UporzadkujPunkty {

    // Insertion Sort: porządkuje tablicę rosnąco, w miejscu.
    static void sortScores(int[] scores) {
        for (int i = 1; i < scores.length; i++) {
            int key = scores[i];
            int j = i - 1;
            while (j >= 0 && scores[j] > key) {
                scores[j + 1] = scores[j];
                j--;
            }
            scores[j + 1] = key;
        }
    }

    public static void main(String[] args) {
        int[] a = {8, 3, 8, -1, 0};
        sortScores(a);
        System.out.println(Arrays.toString(a));   // [-1, 0, 3, 8, 8]

        int[] b = {7};
        sortScores(b);
        System.out.println(Arrays.toString(b));   // [7]

        int[] c = {};
        sortScores(c);
        System.out.println(Arrays.toString(c));   // []

        int[] d = {1000, -1000, 5, 5, -1000};
        sortScores(d);
        System.out.println(Arrays.toString(d));   // [-1000, -1000, 5, 5, 1000]
    }
}

Колоквиум A · Insertion Sort + перевернуть вторую половину
import java.util.Arrays;

public class EgzaminA {

    // Insertion Sort (wykład 04): rosnąco, w tej samej tablicy.
    static void insertionSort(int[] a) {
        for (int i = 1; i < a.length; i++) {
            int key = a[i];
            int j = i - 1;
            while (j >= 0 && a[j] > key) {
                a[j + 1] = a[j];
                j--;
            }
            a[j + 1] = key;
        }
    }

    // Odwraca drugą połowę tablicy: indeksy od a.length / 2 do końca.
    static void reverseSecondHalf(int[] a) {
        int start = a.length / 2;
        int end = a.length - 1;
        while (start < end) {
            int temp = a[start];
            a[start] = a[end];
            a[end] = temp;
            start++;
            end--;
        }
    }

    public static void main(String[] args) {
        int[] arr = {5, 2, 9, 1, 5, 6};
        insertionSort(arr);
        System.out.println(Arrays.toString(arr));   // [1, 2, 5, 5, 6, 9]
        reverseSecondHalf(arr);
        System.out.println(Arrays.toString(arr));   // [1, 2, 5, 9, 6, 5]

        int[] odd = {3, 1, 2, 5, 4};
        insertionSort(odd);
        reverseSecondHalf(odd);
        System.out.println(Arrays.toString(odd));   // [1, 2, 5, 4, 3]

        int[] empty = {};
        insertionSort(empty);
        reverseSecondHalf(empty);
        System.out.println(Arrays.toString(empty)); // []

        int[] one = {7};
        insertionSort(one);
        reverseSecondHalf(one);
        System.out.println(Arrays.toString(one));   // [7]
    }
}

import java.util.Arrays;
import java.util.Random;

public class EgzaminB {

    // n losowych liczb parzystych z przedziału [-100, 100].
    static int[] randomEven(int n) {
        Random rnd = new Random();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = rnd.nextInt(101) * 2 - 100;   // 0..100 -> 0..200 -> -100..100, zawsze parzyste
        }
        return arr;
    }

    // Ile liczb dzieli się przez 4.
    static int countDivisibleBy4(int[] arr) {
        int count = 0;
        for (int i = 0; i < arr.length; i++) {
            if (arr[i] % 4 == 0) {
                count++;
            }
        }
        return count;
    }

    // Nowa tablica z pierwszą połową elementów.
    static int[] firstHalf(int[] arr) {
        int half = arr.length / 2;
        int[] result = new int[half];
        for (int i = 0; i < half; i++) {
            result[i] = arr[i];
        }
        return result;
    }

    public static void main(String[] args) {
        // test na stałych danych
        int[] fixed = {-100, -6, 0, 4, 10, 12, -8};
        System.out.println(countDivisibleBy4(fixed));             // 5  (-100, 0, 4, 12, -8)
        System.out.println(Arrays.toString(firstHalf(fixed)));   // [-100, -6, 0]
        System.out.println(countDivisibleBy4(new int[]{}));       // 0
        System.out.println(Arrays.toString(firstHalf(new int[]{9}))); // []

        // właściwe zadanie: dane losowe
        int[] arr = randomEven(10);
        System.out.println(Arrays.toString(arr));
        System.out.println(countDivisibleBy4(arr));
        System.out.println(Arrays.toString(firstHalf(arr)));
    }
}

⚠️ Ловушки

  • i <= arr.length → ArrayIndexOutOfBoundsException. Правильно i < arr.length.
  • Число вместо переменной: i < 4, arr[3 + i], return 5 → работает только на одном примере.
  • i вместо arr[i] в условии.
  • & вместо %; a <= x <= b не компилируется.
  • Нечётное через % 2 == 1 — ломается на отрицательных.
  • min = 0 вместо min = arr[0].
  • 52 / 10 = 5, не 5.2.
  • Рекурсия: аргумент не уменьшается; результат вызова не используется; неправильная база (у «количества цифр» база 1).
  • Binary search: left = mid (вечный цикл), right = a.length с <=, mid не пересчитан, неотсортированный массив.
  • Insertion: забыли j >= 0; >= вместо > ломает стабильность.
  • Bubble: остановка только после прохода без замен.
  • По убыванию: меняется знак сравнения, а не i++ → i--.
  • println(sortScores(a)) для void — не компилируется.
  • Вторая половина при нечётной длине: длина arr.length - half, не half. Неоднозначность «куда середина» — написать в комментарии, как понял.
  • Готовые Arrays.sort / Arrays.binarySearch запрещены, если сказано «samodzielnie».
